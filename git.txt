# 1. 初始化本地 Git 仓库
git init

# 2. 将源码与 .gitignore 提交至暂存区
git add .

# 3. 提交历史快照并写明修改信息
git commit -m "feat: initial commit for Arduino project"

# 1. 统一主分支名称为 main
git branch -M main

# 2. 关联远端 SSH 地址（替换为你刚复制的真实地址（SSH的））
git remote add origin git@github.com:你的用户名/arduino-blink-demo.git

# 3. 首次推送并建立本地与远端分支追踪
git push -u origin main
